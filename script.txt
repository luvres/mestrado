# Os 6 cenários da dissertação, nos três níveis.
#
# Os três produzem arquivos de amostras IDÊNTICOS byte a byte, e o
# compara_corrida.py confronta cada um com as curvas extraídas da dissertação.
#
# Antes de rodar:
#   C      o binário tem de estar atualizado em relação ao fonte:
#            gcc --std=gnu89 -Wall V8_Reatividade_SimuladorIgor_linux.c -o simulador -lm
#   API    os dois serviços têm de estar no ar:
#            podman-compose up -d
#
# Nos blocos da API, COLE O BLOCO INTEIRO de uma vez. Rodando linha por linha o
# $SID não sobrevive de um comando para o outro, a URL vira /expert//executar,
# o servidor responde 404 e o arquivo sai com só o cabeçalho.

# O expert (8009) decide, o reactor (8008) simula. O arquivo de amostras é
# gravado pelo reactor em /runs, que é ./runs montado do host.
#
# O `| jq -c` no executar não é enfeite: sem ele um 404 ou 502 passa
# despercebido e você só descobre pelo arquivo vazio.

# Cenário 1
rm -f runs/Modelagem_Reator_cenario1.txt

SID=$(curl -s -X POST localhost:8009/expert/initialize \
  -H 'Content-Type: application/json' \
  -d '{"reactor_url":"http://reactor-simulator:8008",
       "pot_turbina":650, "cboro":1800, "posicao_barra":210, "tempodilute":4,
       "t1":-0.3, "t2":-0.8, "delta_i":0, "tmax":1440,
       "arquivo":"Modelagem_Reator_cenario1.txt"}' | jq -r .sessao)

curl -s -X POST localhost:8009/expert/$SID/executar \
  -H 'Content-Type: application/json' -d '{}' | jq -c '{ciclos, t: .estado.t}'

curl -s -X DELETE localhost:8009/expert/$SID > /dev/null

python scripts/compara_corrida.py runs/Modelagem_Reator_cenario1.txt 1

# Cenário 6 — ciclo completo até Cboro<8, ~12 min
# Sem tmax: quem encerra é a regra R19. São ~514 mil ciclos e mais de 400 MB de
# saída, que não cabem numa resposta — daí o laço. O servidor limita cada
# resposta a 8 MB e sinaliza pode_continuar; foram 55 chamadas na última corrida.
rm -f runs/Modelagem_Reator_cenario6.txt

SID=$(curl -s -X POST localhost:8009/expert/initialize \
  -H 'Content-Type: application/json' \
  -d '{"reactor_url":"http://reactor-simulator:8008",
       "pot_turbina":650, "cboro":1800, "posicao_barra":210, "tempodilute":4,
       "t1":-0.3, "t2":-0.8, "delta_i":0,
       "arquivo":"Modelagem_Reator_cenario6.txt"}' | jq -r .sessao)

while [ "$(curl -s -X POST localhost:8009/expert/$SID/executar \
             -H 'Content-Type: application/json' -d '{}' | jq -r .pode_continuar)" = "true" ]; do
  :
done

curl -s localhost:8009/expert/$SID/estado | jq -c '{ciclos, t: .estado.t, Cboro: .estado.Cboro, termino}'
curl -s -X DELETE localhost:8009/expert/$SID > /dev/null

python scripts/compara_corrida.py runs/Modelagem_Reator_cenario6.txt 6
